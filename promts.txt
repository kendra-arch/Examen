Principio de Abierto/Cerrado (OCP - SOLID): ProductoDeportivo debe plantearse como una clase abstracta. Si más adelante necesitas agregar equipamiento (como pesas o balones), simplemente crearás una nueva clase que herede de la abstracta, sin tener que alterar el código que ya funciona.

import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn

import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import androidx.navigation.compose.NavHost
import androidx.navigation.compose.composable
import androidx.navigation.compose.rememberNavController

@Composable
fun AppTiendaDeportiva() {
    // 1. Inicializamos datos de prueba
    val inventario = remember {
        GestorInventario().apply {
            registrarProducto(Calzado(1, "Tenis Running X", 450.0, 10, 42.0, "Running"))
            registrarProducto(Calzado(2, "Botines Fútbol Y", 550.0, 5, 40.5, "Fútbol"))
            registrarProducto(Ropa(3, "Camiseta Térmica", 120.0, 20, "M", "Poliéster"))
            registrarProducto(Ropa(4, "Pantalón Buzo", 180.0, 15, "L", "Algodón"))
        }
    }

    // 2. Configuración de Navegación
    val navController = rememberNavController()

    NavHost(navController = navController, startDestination = "pantalla_calzado") {
        
        // PANTALLA 1: Lista de Calzados
        composable(route = "pantalla_calzado") {
            val calzados = inventario.obtenerInventario().filterIsInstance<Calzado>()
            PantallaListado(
                titulo = "Catálogo de Calzados",
                productos = calzados,
                onNavigate = { navController.navigate("pantalla_ropa") },
                textoBotonNavegacion = "Ver Ropa"
            )
        }

        // PANTALLA 2: Lista de Ropa
        composable(route = "pantalla_ropa") {
            val ropa = inventario.obtenerInventario().filterIsInstance<Ropa>()
            PantallaListado(
                titulo = "Catálogo de Ropa",
                productos = ropa,
                onNavigate = { navController.navigate("pantalla_calzado") },
                textoBotonNavegacion = "Ver Calzados"
            )
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun PantallaListado(
    titulo: String,
    productos: List<ProductoDeportivo>,
    onNavigate: () -> Unit,
    textoBotonNavegacion: String
) {
    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text(titulo) },
                colors = TopAppBarDefaults.topAppBarColors(
                    containerColor = MaterialTheme.colorScheme.primaryContainer
                )
            )
        },
        bottomBar = {
            Button(
                onClick = onNavigate,
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(16.dp)
            ) {
                Text(textoBotonNavegacion)
            }
        }
    ) { paddingValues ->
        // Renderizado dinámico de la lista
        LazyColumn(
            modifier = Modifier
                .padding(paddingValues)
                .fillMaxSize()
                .padding(horizontal = 16.dp)
        ) {
            items(productos, key = { it.id }) { producto ->
                TarjetaProducto(producto)
            }
        }
    }
}

@Composable
fun TarjetaProducto(producto: ProductoDeportivo) {
    Card(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 8.dp),
        elevation = CardDefaults.cardElevation(defaultElevation = 4.dp)
    ) {
        Column(modifier = Modifier.padding(16.dp)) {
            Text(text = producto.nombre, style = MaterialTheme.typography.titleMedium, fontWeight = FontWeight.Bold)
            Text(text = "Precio: Bs. ${producto.precio}")
            Text(text = "Stock: ${producto.stock} unidades")
            
            // Renderizado condicional dependiendo del tipo de objeto
            when (producto) {
                is Calzado -> {
                    Text(text = "Talla: ${producto.talla} | Tipo: ${producto.tipo}")
                }
                is Ropa -> {
                    Text(text = "Talla: ${producto.talla} | Material: ${producto.material}")
                }
            }
        }
    }
}

A. Principio de Responsabilidad Única (SRP - Single Responsibility Principle)

Concepto: Una clase debe tener una, y solo una, razón para cambiar.

Aplicación en tu código: Fíjate que la clase Calzado solo guarda datos de un zapato. No sabe cómo guardarse en una base de datos ni cómo sumarse a una lista. Para eso creamos GestorInventario.
